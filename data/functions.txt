sum sum(a, b) = a + b
sub sub(a, b) = a - b
mul mul(a, b) = a * b
div div(a, b) = a / b (целочисленное деление a на b)
mod mod(a, b) = a % b (остаток от a при делении на b)
abs abs(a, b) = |a|
pow pow(a, b) = a ^ b (возведение a в степень b)
rand rand(a, b) = случайное целое число от a до b (включительно)
and and(a, b) = a & b (побитовое И)
or or(a, b) = a | b (побитовое ИЛИ)
not not(a, b) = ~a (побитовое НЕ)
xor xor(a, b) = a ^ b (побитовое исключающее ИЛИ)
avg avg(a, b) = (a + b) / 2 (среднее арифметическое, округленное вниз)
sumr sumr(a, b) = a + b (с погрешностью +-10)
subr subr(a, b) = a - b (с погрешностью +-10)
mulr mulr(a, b) = a * b (с погрешностью +-25)
divr divr(a, b) = a / b (с погрешностью +-25)
conc conc(a, b) = ab (конкатенация, соединение чисел воедино)
base base(a, b) = перевод числа a из b-ичной системы счисления в десятичную
round round(a, b) = округление числа a до b десятичных разрядов
five five(a, b) = 5 (постоянная функция, игнорирует аргументы)
neg2 neg2(a, b) = -a + 2 (линейная функция от первого аргумента)
